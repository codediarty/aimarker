입점사 상세페이지 일괄 검수(예)
import { AIMarker } from './aimarker.mjs';
const am = new AIMarker({ key: process.env.AIMARKER_KEY });

const items = products.map((p) => ({ id: p.id, text: p.copy }));
const { results } = await am.batch(items);       // 20건씩

for (const r of results) {
  if (r.decision === 'clear') publish(r.id);     // 바로 게시
  else if (r.decision === 'expert') review(r.id); // 심사 담당자
  else askSeller(r.id, r.findings);              // 바꿀 말과 함께
}
